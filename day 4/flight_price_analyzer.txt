CLASS Flight
    ATTRIBUTES: flightId, price, duration

FUNCTION customComparator(flight1, flight2)
    IF flight1.price != flight2.price
        RETURN flight1.price < flight2.price
    ELSE
        RETURN flight1.duration < flight2.duration

FUNCTION mergeSort(flights, comparator)
    IF length(flights) <= 1
        RETURN flights
    mid = length/2
    left = mergeSort(flights[0..mid], comparator)
    right = mergeSort(flights[mid+1..end], comparator)
    RETURN merge(left, right, comparator)

FUNCTION merge(left, right, comparator)
    result = []
    WHILE left and right not empty
        IF comparator(left[0], right[0]) is TRUE
            result.add(left.pop(0))
        ELSE
            result.add(right.pop(0))
    APPEND remaining left or right
    RETURN result

MAIN
    flights = [list of Flight]
    sortedFlights = mergeSort(flights, customComparator)
    DISPLAY sortedFlights
